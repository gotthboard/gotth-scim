package scim // import "github.com/gotthboard/gotth-scim/pkg/scim"

Package scim implements a storage-neutral SCIM 2.0 HTTP server, protocol
validation, transactional storage contract, tombstones, and reconciliation.
Applications own authentication, durable adapters, and product policy.

FUNCTIONS

func ApplyPatch(definition ResourceDefinition, current Document, request PatchRequest, resourceID string) (Document, []IndexKey, string, error)
    ApplyPatch applies an RFC 7644 PATCH request to a private copy of current
    and returns a fully revalidated replacement document.

func CheckStore(ctx context.Context, factory func() Store) error
    CheckStore verifies the transaction, identity, copy, isolation, uniqueness,
    deletion, and tombstone requirements of a fresh Store implementation.

func CoreKeyCases() map[string]string
    CoreKeyCases is the protocol key set used by User, Group, PATCH, and Bulk
    messages. Extensions can pass an augmented copy to NormalizeKeys.

func IfMatch(header, current string, exists bool) (bool, error)
    IfMatch reports whether an If-Match value admits mutation of a resource with
    the current strong version. An empty header means no precondition.

func IfNoneMatch(header, current string, exists bool) (bool, error)
    IfNoneMatch reports whether a request may proceed. For reads, false maps to
    304. For writes, false maps to 412. Weak comparison is intentional.

func MatchFilter(expression *FilterExpression, document Document) (bool, error)
    MatchFilter evaluates a validated filter against one canonical resource.

func NewResourceID(entropy io.Reader) (string, error)
    NewResourceID returns one opaque 128-bit base64url identifier. The caller
    must persist it; regenerating IDs from mutable names violates SCIM identity.

func NormalizeKeys(value any, canonical map[string]string) (any, error)
    NormalizeKeys recursively canonicalizes known SCIM attribute spelling and
    rejects case-equivalent collisions instead of choosing one attacker-supplied
    value.

func ParseBulkPath(raw string) (collection, resourceID, bulkReference string, err error)
    ParseBulkPath admits one relative collection or collection/resource path.
    Collection support is determined later by the configured Registry. A bulkId
    reference is returned separately and must be resolved only from a prior
    successful operation.

func ParseEqualityFilter(raw, supportedAttribute string) (string, error)
    ParseEqualityFilter accepts the deliberately bounded interoperable subset
    `<attribute> eq "<value>"` for one caller-selected attribute.

func ValidateSchemas(actual, expected []string) error
    ValidateSchemas requires an exact case-insensitive schema set with no
    duplicates. Extra schemas are not silently ignored.

func Version(canonical []byte) (string, error)
    Version returns a strong quoted SHA-256 entity tag for one canonical
    resource representation. Callers must include every state field whose change
    should invalidate conditional requests.

type AuthenticationScheme struct {
	Type             string `json:"type"`
	Name             string `json:"name"`
	Description      string `json:"description"`
	SpecURI          string `json:"specUri,omitempty"`
	DocumentationURI string `json:"documentationUri,omitempty"`

    AuthenticationScheme documents authentication performed before Server.
    Server never parses credentials itself.

type BulkOperation struct {
	Method  string          `json:"method"`

	Path    string          `json:"path"`
	Version string          `json:"version,omitempty"`

    BulkOperation is one independently transactional SCIM operation.

type BulkRequest struct {
	Schemas      []string        `json:"schemas"`

	Operations   []BulkOperation `json:"Operations"`
}
    BulkRequest is the bounded RFC 7644 bulk envelope.

func DecodeBulk(raw []byte) (BulkRequest, error)
    DecodeBulk performs bounded strict decoding and generic method/path/bulkId
    validation without executing or partially committing anything.

type BulkResponse struct {
	Schemas    []string                `json:"schemas"`

type DesiredResource struct {
	ResourceType string
	ExternalID   string

    DesiredResource is one manager-owned desired SCIM resource.

type Document map[string]any
    Document is one decoded SCIM resource body. Callers must treat values
    returned by this package as immutable.

func DecodeDocument(raw []byte) (Document, error)
    DecodeDocument strictly decodes one bounded JSON object. Unlike ordinary
    encoding/json map decoding, it rejects duplicate and case-equivalent names.

type ErrorResponse struct {
	Schemas  []string `json:"schemas"`
	Status   string   `json:"status"`
	SCIMType string   `json:"scimType,omitempty"`
	Detail   string   `json:"detail"`
}
    ErrorResponse is the RFC 7644 error envelope.

func NewError(status int, scimType, detail string) (ErrorResponse, error)
    NewError creates a canonical SCIM error without logging or exposing an
    underlying implementation error.

type Extension struct {
	Schema      string
	Name        string
	Description string
	Attributes  []SchemaAttribute
	Required    bool
	Validate    func(Document) error

    Extension describes one admitted schema extension. Validate receives the

func EnterpriseUserExtension() Extension
    EnterpriseUserExtension returns the optional standard RFC 7643 enterprise
    User extension. It is included, but not required, by DefaultDefinitions.

	// Has unexported fields.

type PasswordStore interface {
	Store
	SupportsPasswordTransactions()
}
    PasswordStore is an explicit capability marker. Servers refuse to advertise
    changePassword unless the configured Store implements it.

type PasswordTransaction interface {
	SetPassword(scope, resourceType, id string, password []byte) (string, error)
}
    PasswordTransaction is the optional atomic credential adapter used when
    changePassword is advertised. It MUST consume but never retain the supplied
    slice, return a non-secret revision token, and commit or roll back with the
    enclosing Store transaction.

type Record struct {
	Scope        string
	ResourceType string
	ID           string
	ExternalID   string

	Data              []byte
	Indexes           []IndexKey

    Record is the storage representation of one live SCIM resource. Data is

type Store interface {
	Transact(context.Context, func(Transaction) error) error
}
    Store runs one callback in one atomic transaction. Implementations must call
    fn exactly once, commit only when fn returns nil, and never retry it.

type Tombstone struct {
	Scope        string
	ResourceType string
	ID           string
	ExternalID   string
	Manager      string
	Version      string
	DeletedAt    time.Time
}
    Tombstone permanently reserves a deleted provider ID and records the
    provisioning ownership needed to prevent unsafe automatic recreation.

type Transaction interface {
	Get(scope, resourceType, id string) (Record, error)
	List(Query) ([]Record, error)
	Create(Record) error
	Replace(Record, string) error
	Delete(scope, resourceType, id, expectedVersion string, tombstone Tombstone) error
	Tombstones(scope, resourceType string) ([]Tombstone, error)
}
    Transaction is the complete persistence contract used by Server and
    Reconciler. Returned values and accepted slices must be defensively copied.

type WriteMode uint8
    WriteMode distinguishes create from replacement validation.

const (
	CreateMode WriteMode = iota + 1
	ReplaceMode
)
